cmake_minimum_required(VERSION 3.0.2)
project(ov13855_apriltag_vision)
add_compile_options(-std=c++14 -Wall -Wextra)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(catkin REQUIRED COMPONENTS cv_bridge geometry_msgs message_generation roscpp sensor_msgs std_msgs tf2_ros)
find_package(OpenCV REQUIRED COMPONENTS core calib3d imgproc)
find_path(APRILTAG_INCLUDE_DIR apriltag.h PATH_SUFFIXES apriltag)
find_library(APRILTAG_LIBRARY apriltag)
if(NOT APRILTAG_INCLUDE_DIR OR NOT APRILTAG_LIBRARY)
  message(FATAL_ERROR "AprilTag development headers/library required")
endif()
add_message_files(FILES HomogeneousTransform.msg)
generate_messages(DEPENDENCIES std_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES ov13855_pose_solver ov13855_apriltag_detector
 CATKIN_DEPENDS cv_bridge geometry_msgs message_runtime roscpp sensor_msgs std_msgs tf2_ros)
include_directories(include ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${APRILTAG_INCLUDE_DIR})
add_library(ov13855_pose_solver src/pose_solver.cpp)
target_link_libraries(ov13855_pose_solver ${OpenCV_LIBRARIES})
add_library(ov13855_apriltag_detector src/apriltag_detector.cpp)
target_link_libraries(ov13855_apriltag_detector ${OpenCV_LIBRARIES} ${APRILTAG_LIBRARY})
add_executable(ov13855_apriltag_vision_node src/ov13855_apriltag_vision_node.cpp)
add_dependencies(ov13855_apriltag_vision_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(ov13855_apriltag_vision_node ov13855_pose_solver ov13855_apriltag_detector ${catkin_LIBRARIES})
install(TARGETS ov13855_apriltag_vision_node ov13855_pose_solver ov13855_apriltag_detector
 RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
 LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch validation DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md VALIDATION.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
 find_package(rostest REQUIRED)
 catkin_add_gtest(test_ov13855_pose_solver test/test_pose_solver.cpp)
 if(TARGET test_ov13855_pose_solver)
  target_link_libraries(test_ov13855_pose_solver ov13855_pose_solver ${OpenCV_LIBRARIES})
 endif()
 catkin_add_gtest(test_apriltag_detector test/test_apriltag_detector.cpp)
 if(TARGET test_apriltag_detector)
  target_link_libraries(test_apriltag_detector ov13855_apriltag_detector ov13855_pose_solver ${OpenCV_LIBRARIES})
 endif()
 add_rostest(test/test_pipeline.test)
endif()
