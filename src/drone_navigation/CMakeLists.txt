cmake_minimum_required(VERSION 3.0.2)
project(drone_navigation)
find_package(catkin REQUIRED COMPONENTS rospy actionlib_msgs geometry_msgs nav_msgs std_msgs message_generation)
catkin_python_setup()
add_action_files(FILES ExecuteRoute.action)
add_message_files(FILES NavigationCommand.msg)
generate_messages(DEPENDENCIES actionlib_msgs geometry_msgs std_msgs)
catkin_package(CATKIN_DEPENDS message_runtime actionlib_msgs geometry_msgs std_msgs)
catkin_install_python(PROGRAMS scripts/nav_inputs scripts/waypoint_manager scripts/trajectory_executor scripts/offboard_gateway scripts/run_route DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
 catkin_add_nosetests(test/test_core.py)
 catkin_add_nosetests(test/test_runtime.py)
endif()
