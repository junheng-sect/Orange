cmake_minimum_required(VERSION 3.0.2)
project(barcode_rc_trigger)
find_package(catkin REQUIRED COMPONENTS barcode_reader_camera barcode_reader_serial mavros_msgs rospy std_msgs)
catkin_python_setup()
catkin_package(CATKIN_DEPENDS barcode_reader_camera barcode_reader_serial mavros_msgs rospy std_msgs)
catkin_install_python(PROGRAMS scripts/barcode_rc_trigger_node
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  catkin_add_nosetests(test/test_policy.py)
  catkin_add_nosetests(test/test_runner.py)
  add_rostest(test/node.test)
endif()
