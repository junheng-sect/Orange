cmake_minimum_required(VERSION 3.10)
project(dual_camera_recorder)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs std_msgs)
find_package(Boost REQUIRED COMPONENTS filesystem system)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil)
catkin_package()
add_executable(dual_camera_recorder_node src/recorder.cpp src/gray_video.cpp)
target_compile_features(dual_camera_recorder_node PRIVATE cxx_std_14)
target_include_directories(dual_camera_recorder_node PRIVATE ${catkin_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS})
target_link_libraries(dual_camera_recorder_node ${catkin_LIBRARIES} ${Boost_LIBRARIES} PkgConfig::FFMPEG)
install(TARGETS dual_camera_recorder_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
