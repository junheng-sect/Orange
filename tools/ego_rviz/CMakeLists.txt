cmake_minimum_required(VERSION 3.10)
project(ego_display LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(catkin REQUIRED COMPONENTS roscpp tf sensor_msgs nav_msgs visualization_msgs)
find_package(Armadillo REQUIRED)
add_executable(odom_visualization src/odom_visualization.cpp vendor/pose_utils/src/pose_utils.cpp)
target_include_directories(odom_visualization PRIVATE vendor/pose_utils/include ${catkin_INCLUDE_DIRS} ${ARMADILLO_INCLUDE_DIRS})
target_link_libraries(odom_visualization ${catkin_LIBRARIES} ${ARMADILLO_LIBRARIES})
set_target_properties(odom_visualization PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/vendor/odom_visualization/bin")
